cmake_minimum_required(VERSION 3.22)
project(FundFuneral VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets Sql Qml Network Test)
add_executable(fund-funeral desktop/main.cpp desktop/engine.cpp desktop/sync.cpp desktop/window.cpp desktop/vendor/qrcodegen.cpp desktop/resources.qrc)
target_link_libraries(fund-funeral PRIVATE Qt6::Widgets Qt6::Sql Qt6::Qml Qt6::Network)
target_compile_options(fund-funeral PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS fund-funeral RUNTIME DESTINATION bin)
enable_testing()
add_test(NAME desktop-self-test COMMAND fund-funeral --self-test)
set_tests_properties(desktop-self-test PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
